cmake_minimum_required(VERSION 3.20)
project(MoonlightRT81 LANGUAGES C CXX RC)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_compile_definitions(_WIN32_WINNT=0x0603 WINVER=0x0603 NTDDI_VERSION=0x06030000 UNICODE _UNICODE)
add_compile_options(-O2 -fstack-protector-strong -ffunction-sections -fdata-sections)
# Keep build-machine usernames and paths out of __FILE__, assertions and debug data.
get_filename_component(RT_COMPILER_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
get_filename_component(RT_COMPILER_ROOT "${RT_COMPILER_BIN}" DIRECTORY)
foreach(RT_MAP "${CMAKE_SOURCE_DIR}=." "${RT_COMPILER_ROOT}=toolchain")
  add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:-ffile-prefix-map=${RT_MAP}>")
endforeach()
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src)
add_compile_definitions(MBEDTLS_USER_CONFIG_FILE="rt_mbedtls_config.h")
add_subdirectory(work/mbedtls work/build-mbedtls)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
add_subdirectory(work/moonlight-common-c work/build-common)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_FIXED_POINT ON CACHE BOOL "" FORCE)
set(OPUS_ENABLE_FLOAT_API OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
add_subdirectory(work/opus work/build-opus)
add_library(rt_xml STATIC work/pugixml/pugixml.cpp)
target_compile_options(rt_xml PRIVATE -fno-exceptions -fno-rtti)
target_compile_definitions(rt_xml PUBLIC PUGIXML_NO_STL PUGIXML_NO_EXCEPTIONS)
target_include_directories(rt_xml PUBLIC work/pugixml)
add_executable(MoonlightRT81 WIN32 src/main.cpp src/ui.cpp src/mouse.cpp src/input_dispatch.cpp src/artwork.cpp src/util.cpp src/host.cpp src/video.cpp src/audio.cpp src/video_hooks.c src/loader_config.c src/app.rc)
target_include_directories(MoonlightRT81 PRIVATE assets)
set_source_files_properties(src/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest;${CMAKE_CURRENT_SOURCE_DIR}/assets/moonlight.ico")
target_compile_options(MoonlightRT81 PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
target_link_options(MoonlightRT81 PRIVATE -nostartfiles -nostdlib++ -static "-Wl,--entry,entry,--subsystem,windows:6.3,--gc-sections,--dynamicbase,--nxcompat,--major-os-version,6,--minor-os-version,3")
target_link_libraries(MoonlightRT81 PRIVATE moonlight-common-c mbedtls mbedx509 mbedcrypto opus rt_xml d3d11 dxgi mfplat mfuuid ole32 shell32 uuid strmiids winmm bcrypt crypt32 ws2_32 gdi32 comctl32)
